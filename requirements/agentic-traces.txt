# Dependencies for the AGENTIC_TRACES venv (Python 3.12): the agentic
# trace-replay benchmark driven by SemiAnalysis's agentx-harness AIPerf fork.
#
# Does NOT include llm-aiperf.txt the way prefix-cache.txt does: the client
# here is the agentx-harness fork below, which ships the same `aiperf` package
# as the released aiperf and would collide with it. llm-aiperf.txt's
# torch==2.4.0+cpu pin is a constraint of the released aiperf, not of the fork
# (which needs no torch at all).
#
# Does NOT apply constraints.txt either: its pillow==11.1 conflicts with the
# fork's pillow~=12.3. The one other entry that matters here, pyjwt, is pinned
# below to the same version.

# SemiAnalysis's AIPerf fork that owns the inferencex-agentx-mvp scenario and
# the Weka trace dataset loaders, so this pin is part of the benchmark
# definition: results are only comparable across runs on the same version.
# 1.0.13 is the release InferenceX's own GPU reference runs install
# (inferencex-e2e/infx/bench/agentic/requirements.txt), and it includes the
# fixes we used to patch in locally (agentx-harness#49, #67).
agentx-harness==1.0.13

# Workflow-engine eager import: workflow_module -> test_module.context ->
# health_tests uses aiohttp at module top.
aiohttp

# JWT minting for a JWT-protected inference server.
pyjwt==2.7.0

# HTTP clients used by the readiness probes and dataset download.
requests
urllib3

# Weka trace datasets are pulled from the Hugging Face Hub. The datasets floor
# matches InferenceX's AgentX requirements.
datasets>=4.7.0
huggingface_hub[cli]>=0.25.0

# Required by models whose HF repo ships a custom tiktoken-based tokenizer
# (e.g. moonshotai/Kimi-*). transformers loads tokenization_kimi.py via
# trust_remote_code, which imports tiktoken (and blobfile to fetch the vocab).
tiktoken
blobfile

# SwarmOne swo-bench replay client (TraceSource.SWARMONE). It replays recorded
# Claude-Code / Codex coding sessions against the endpoint; the replay plan is
# built server-side by the SwarmOne backend. A SwarmOne license (SWO_LICENSE_KEY / ~/.swarmone/license.key) is
# required at run time, checked up front in run.py.
#
# Deliberately unpinned: SwarmOne keeps only the latest release on PyPI and
# deletes the previous ones, so a version pin stops resolving the moment they
# publish again. The installed version is recorded per run (swo_bench_version
# in the payload) instead.
swo-bench
