# Dependencies for the SPEC_DECODE workflow venv (Python 3.11).
# Used by direct invocations of run_workflows.py when
# --spec-decode is set (via the run_spec_decode.py launcher).
#
# Mirrors requirements/benchmarks-spec-decode.txt (the v1 spec-decode
# venv) plus aiohttp for the v2 orchestrator's eager import chain
# (workflow_module -> test_module.context -> health_tests). Like its v1
# counterpart, this file intentionally omits `-c constraints.txt`:
# aiperf >= 0.8.0 (needed for the Spec-Bench / SPEED-Bench dataset
# plugins) requires pillow >= 12.2, which conflicts with the shared
# pillow==11.1 pin in constraints.txt. pyjwt is pinned explicitly to
# keep cross-venv alignment for the JWT auth path.

aiperf==0.8.0

# only transformers 5.x parses; under 4.x AIPerf's tokenizer load fails with
#   AttributeError: 'list' object has no attribute 'keys'
# before any request is sent (same issue as llm-vllm-overrides.txt). AIPerf only
# tokenizes and does not need torch, so the old torch==2.4.0+cpu pin, which
# capped transformers at <5, is dropped.
transformers>=5.5.0,<6
pyjwt==2.7.0
